<script>
    const TICKET_PRICE = 250;

    // أرقام الواتساب الخاصة بكل مسؤول حسب طريقة الدفع (استبدلها بالأرقام الحقيقية)
    const WHATSAPP_NUMBERS = {
        mekhail: {
            instapay: '201000000001',
            vf:       '201000000001',
            orange:   '201200000001'
        },
        fayez: {
            instapay: '201000000002',
            vf:       '201000000002',
            orange:   '201200000002'
        }
    };

    const LABELS = {
        recipients: { mekhail: 'ميخائيل إدور', fayez: 'فايز فيكتور' },
        payments: { instapay: 'إنستا باي', vf: 'فودافون كاش', orange: 'أورانج كاش' },
        colors: { red: 'أحمر', blue: 'أزرق', yellow: 'أصفر', green: 'أخضر' },
        buses: { bus1: 'الباص الأول', bus2: 'الباص الثاني' }
    };

    let registrations = JSON.parse(localStorage.getItem('trip_registrations')) || [];

    // تحويل أي بيانات قديمة مخزنة في المتصفح إلى الهيكل الجديد (members) تلقائياً
    function migrateOldData() {
        let changed = false;
        registrations = registrations.map(reg => {
            if (!reg.members) {
                changed = true;
                const isPaid = reg.status === 'تم التأكيد';
                const members = [{
                    id: `${reg.id}_0`,
                    name: reg.mainAttendee?.name || 'غير معروف',
                    phone: reg.mainAttendee?.phone || '',
                    isMain: true,
                    paid: isPaid,
                    cancelled: false
                }];
                (reg.companions || []).forEach((c, idx) => {
                    members.push({
                        id: `${reg.id}_${idx + 1}`,
                        name: c.name,
                        phone: c.phone,
                        isMain: false,
                        paid: isPaid,
                        cancelled: false
                    });
                });
                reg.members = members;
            }
            return reg;
        });
        if (changed) saveRegistrations();
    }

    function saveRegistrations() {
        localStorage.setItem('trip_registrations', JSON.stringify(registrations));
    }

    document.getElementById('registrationForm').addEventListener('input', calculateTotal);
    document.getElementById('registrationForm').addEventListener('submit', handleFormSubmit);

    function toggleCompanions() {
        const isGroup = document.querySelector('input[name="regType"]:checked').value === 'group';
        const section = document.getElementById('companions-section');
        if (isGroup) {
            section.classList.remove('hidden');
            generateCompanionFields();
        } else {
            section.classList.add('hidden');
            document.getElementById('companionFields').innerHTML = '';
        }
        calculateTotal();
    }

    function generateCompanionFields() {
        const count = parseInt(document.getElementById('companionCount').value);
        const container = document.getElementById('companionFields');
        container.innerHTML = '';
        for (let i = 1; i <= count; i++) {
            container.innerHTML += `
                <div class="companion-entry">
                    <h4>المرافق ${i}</h4>
                    <div class="form-group">
                        <label>الاسم الثلاثي:</label>
                        <input type="text" class="comp-name" required placeholder="اسم المرافق">
                    </div>
                    <div class="form-group">
                        <label>رقم الهاتف:</label>
                        <input type="tel" class="comp-phone" required placeholder="رقم هاتف المرافق">
                    </div>
                </div>
            `;
        }
        calculateTotal();
    }

    function calculateTotal() {
        const isGroup = document.querySelector('input[name="regType"]:checked').value === 'group';
        let totalPersons = 1;
        if (isGroup) {
            totalPersons += parseInt(document.getElementById('companionCount').value);
        }
        document.getElementById('totalPersonsDisplay').innerText = totalPersons;
        document.getElementById('totalPrice').innerText = totalPersons * TICKET_PRICE;
    }

    function handleFormSubmit(e) {
        e.preventDefault();
        const isGroup = document.querySelector('input[name="regType"]:checked').value === 'group';
        const regId = Date.now();

        const members = [{
            id: `${regId}_0`,
            name: document.getElementById('fullName').value.trim(),
            phone: document.getElementById('phone').value.trim(),
            isMain: true,
            paid: false,
            cancelled: false
        }];

        if (isGroup) {
            const names = document.getElementsByClassName('comp-name');
            const phones = document.getElementsByClassName('comp-phone');
            for (let i = 0; i < names.length; i++) {
                members.push({
                    id: `${regId}_${i + 1}`,
                    name: names[i].value.trim(),
                    phone: phones[i].value.trim(),
                    isMain: false,
                    paid: false,
                    cancelled: false
                });
            }
        }

        const recipient = document.querySelector('input[name="recipient"]:checked').value;
        const paymentMethod = document.getElementById('paymentMethod').value;

        const registrationData = {
            id: regId,
            isGroup,
            members,
            bus: document.querySelector('input[name="busChoice"]:checked').value,
            colorGroup: document.querySelector('input[name="colorGroup"]:checked').value,
            recipient,
            paymentMethod,
            timestamp: new Date().toLocaleString('ar-EG')
        };

        registrations.push(registrationData);
        saveRegistrations();

        redirectToWhatsApp(registrationData);

        document.getElementById('registrationForm').reset();
        toggleCompanions();

        if (document.getElementById('admin-panel').style.display === 'block') {
            renderAdminPanel();
        }
    }

    function redirectToWhatsApp(data) {
        const phoneNum = WHATSAPP_NUMBERS[data.recipient][data.paymentMethod];
        const recipientName = LABELS.recipients[data.recipient];
        const paymentName = LABELS.payments[data.paymentMethod];
        const mainMember = data.members[0];
        const companions = data.members.slice(1);
        const totalPersons = data.members.length;
        const totalPrice = totalPersons * TICKET_PRICE;

        const companionsNames = companions.length
            ? `\nالمرافقون: ${companions.map(c => c.name).join('، ')}`
            : '';

        const message = `مرحباً ${recipientName}، أود تأكيد حجز رحلة دير الأنبا أنطونيوس.\n` +
            `الاسم: ${mainMember.name}\n` +
            `عدد الأفراد: ${totalPersons}${companionsNames}\n` +
            `الإجمالي: ${totalPrice} جنيه\n` +
            `طريقة الدفع: ${paymentName}\n` +
            `الباص: ${LABELS.buses[data.bus]} | المجموعة: ${LABELS.colors[data.colorGroup]}\n` +
            `كود الحجز: ${data.id}`;

        window.open(`https://wa.me/${phoneNum}?text=${encodeURIComponent(message)}`, '_blank');
    }

    // --- حسابات الحجز الواحد ---
    function getRegSummary(reg) {
        let activeCount = 0;
        let paidCount = 0;
        let cancelledCount = 0;

        reg.members.forEach(m => {
            if (m.cancelled) {
                cancelledCount++;
            } else {
                activeCount++;
                if (m.paid) paidCount++;
            }
        });

        const unpaidCount = activeCount - paidCount;
        let statusKey = 'unpaid';
        if (activeCount === 0) statusKey = 'cancelled';
        else if (paidCount === activeCount) statusKey = 'paid';
        else if (paidCount > 0) statusKey = 'partial';

        return {
            activeCount,
            paidCount,
            unpaidCount,
            cancelledCount,
            expectedMoney: activeCount * TICKET_PRICE,
            paidMoney: paidCount * TICKET_PRICE,
            remainingMoney: unpaidCount * TICKET_PRICE,
            statusKey
        };
    }

    // --- إجراءات التحكم في الأفراد والمجموعات ---
    function toggleMemberPaid(regId, memberId) {
        const reg = registrations.find(r => r.id === regId);
        if (!reg) return;
        const member = reg.members.find(m => m.id === memberId);
        if (member && !member.cancelled) {
            member.paid = !member.paid;
            saveRegistrations();
            renderAdminPanel();
        }
    }

    function toggleMemberCancelled(regId, memberId) {
        const reg = registrations.find(r => r.id === regId);
        if (!reg) return;
        const member = reg.members.find(m => m.id === memberId);
        if (member) {
            member.cancelled = !member.cancelled;
            if (member.cancelled) member.paid = false; // الملغي لا يُحسب مدفوعاً
            saveRegistrations();
            renderAdminPanel();
        }
    }

    function approveAllInReg(regId) {
        const reg = registrations.find(r => r.id === regId);
        if (!reg) return;
        reg.members.forEach(m => {
            if (!m.cancelled) m.paid = true;
        });
        saveRegistrations();
        renderAdminPanel();
    }

    function cancelAllInReg(regId) {
        const reg = registrations.find(r => r.id === regId);
        if (!reg) return;
        const allCancelled = reg.members.every(m => m.cancelled);
        reg.members.forEach(m => {
            m.cancelled = !allCancelled;
            if (m.cancelled) m.paid = false;
        });
        saveRegistrations();
        renderAdminPanel();
    }

    // --- منطق لوحة التحكم والحسابات ---
    function checkAdminAccess() {
        migrateOldData();
        const urlParams = new URLSearchParams(window.location.search);
        if (urlParams.get('admin') === 'true') {
            document.getElementById('admin-panel').style.display = 'block';
            renderAdminPanel();
        }
    }

    function calculateStats(filterRecipientKey = null) {
        const list = filterRecipientKey
            ? registrations.filter(r => r.recipient === filterRecipientKey)
            : registrations;

        const stats = {
            expectedMoney: 0,
            paidMoney: 0,
            remainingMoney: 0,
            activePersons: 0,
            paidPersons: 0,
            cancelledPersons: 0,
            byPayment: {
                instapay: { expected: 0, paid: 0, persons: 0 },
                vf:       { expected: 0, paid: 0, persons: 0 },
                orange:   { expected: 0, paid: 0, persons: 0 }
            }
        };

        list.forEach(r => {
            const s = getRegSummary(r);
            stats.expectedMoney += s.expectedMoney;
            stats.paidMoney += s.paidMoney;
            stats.remainingMoney += s.remainingMoney;
            stats.activePersons += s.activeCount;
            stats.paidPersons += s.paidCount;
            stats.cancelledPersons += s.cancelledCount;

            if (stats.byPayment[r.paymentMethod]) {
                stats.byPayment[r.paymentMethod].expected += s.expectedMoney;
                stats.byPayment[r.paymentMethod].paid += s.paidMoney;
                stats.byPayment[r.paymentMethod].persons += s.activeCount;
            }
        });

        return stats;
    }

    function renderAdminTable() {
        const recFilter = document.getElementById('filterRecipient').value;
        const payFilter = document.getElementById('filterPayment').value;
        const statusFilter = document.getElementById('filterStatus').value;

        const filtered = registrations.filter(reg => {
            const matchRec = recFilter === 'all' || reg.recipient === recFilter;
            const matchPay = payFilter === 'all' || reg.paymentMethod === payFilter;
            const summary = getRegSummary(reg);
            const matchStatus = statusFilter === 'all' || summary.statusKey === statusFilter;
            return matchRec && matchPay && matchStatus;
        });

        const tbody = document.getElementById('adminTableBody');
        tbody.innerHTML = '';

        if (filtered.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" style="text-align:center;">لا توجد بيانات مطابقة للفرز الحالي</td></tr>';
            return;
        }

        filtered.forEach(reg => {
            const row = tbody.insertRow();
            const s = getRegSummary(reg);
            const allCancelled = s.activeCount === 0;

            const membersHtml = reg.members.map(m => {
                const rowClass = m.cancelled ? 'is-cancelled' : (m.paid ? 'is-paid' : '');
                const badge = m.cancelled
                    ? '<span class="badge badge-cancelled">ملغي</span>'
                    : (m.paid ? '<span class="badge badge-paid">مدفوع</span>' : '<span class="badge badge-pending">لم يدفع</span>');

                const payBtn = m.cancelled ? '' : `
                    <button type="button" class="btn-mini ${m.paid ? 'btn-unpay' : 'btn-approve'}"
                        onclick="toggleMemberPaid(${reg.id}, '${m.id}')">
                        ${m.paid ? 'إلغاء الدفع' : 'تأكيد دفع ✓'}
                    </button>
                `;

                const cancelBtn = `
                    <button type="button" class="btn-mini ${m.cancelled ? 'btn-restore' : 'btn-cancel'}"
                        onclick="toggleMemberCancelled(${reg.id}, '${m.id}')">
                        ${m.cancelled ? 'استرجاع' : 'إلغاء ✕'}
                    </button>
                `;

                return `
                    <div class="member-row ${rowClass}">
                        <div class="member-info">
                            <span class="member-name"><strong>${m.name}</strong> ${m.isMain ? '(الرئيسي)' : ''}</span>
                            <div style="font-size:0.85em; color:#555;">${m.phone} ${badge}</div>
                        </div>
                        <div class="member-actions">
                            ${payBtn}
                            ${cancelBtn}
                        </div>
                    </div>
                `;
            }).join('');

            row.innerHTML = `
                <td style="min-width: 300px;">${membersHtml}</td>
                <td>
                    <div><strong>${LABELS.buses[reg.bus] || reg.bus}</strong></div>
                    <div>المجموعة: ${LABELS.colors[reg.colorGroup] || reg.colorGroup}</div>
                    <div style="font-size:0.8em; color:#777; margin-top:4px;">${reg.isGroup ? 'مجموعة' : 'فردي'}</div>
                </td>
                <td>
                    <div><strong>${LABELS.recipients[reg.recipient] || 'غير محدد'}</strong></div>
                    <div>${LABELS.payments[reg.paymentMethod] || reg.paymentMethod}</div>
                    <div style="font-size:0.78em; color:#777; margin-top:4px;">${reg.timestamp}</div>
                </td>
                <td style="white-space: nowrap;">
                    <div>المطلوب: <strong>${s.expectedMoney} ج</strong> (${s.activeCount} نشط)</div>
                    <div style="color:#27ae60;">المدفوع: <strong>${s.paidMoney} ج</strong> (${s.paidCount} فرد)</div>
                    <div style="color:#e67e22;">المتبقي: <strong>${s.remainingMoney} ج</strong></div>
                </td>
                <td>
                    <div class="group-actions">
                        ${!allCancelled ? `<button type="button" class="btn-mini btn-approve" onclick="approveAllInReg(${reg.id})">Approve الكل ✓</button>` : ''}

                        </button>
                    </div>
                </td>
            `;
        });
    }

    function exportData() {
        const recFilter = document.getElementById('filterRecipient').value;
        const payFilter = document.getElementById('filterPayment').value;
        const statusFilter = document.getElementById('filterStatus').value;

        const filtered = registrations.filter(reg => {
            const matchRec = recFilter === 'all' || reg.recipient === recFilter;
            const matchPay = payFilter === 'all' || reg.paymentMethod === payFilter;
            const matchStatus = statusFilter === 'all' || getRegSummary(reg).statusKey === statusFilter;
            return matchRec && matchPay && matchStatus;
        });

        let csvContent = "\uFEFFكود الحجز,الاسم,الهاتف,الصفة,حالة الفرد,الباص,المجموعة,المستلم,طريقة الدفع,المبلغ المدفوع للفرد,التاريخ\n";
        filtered.forEach(r => {
            r.members.forEach(m => {
                const statusText = m.cancelled ? 'ملغي' : (m.paid ? 'مدفوع' : 'لم يدفع');
                const paidAmount = (!m.cancelled && m.paid) ? TICKET_PRICE : 0;
                const row = [
                    r.id,
                    m.name,
                    m.phone,
                    m.isMain ? 'رئيسي' : 'مرافق',
                    statusText,
                    LABELS.buses[r.bus] || r.bus,
                    LABELS.colors[r.colorGroup] || r.colorGroup,
                    LABELS.recipients[r.recipient] || '',
                    LABELS.payments[r.paymentMethod] || r.paymentMethod,
                    paidAmount,
                    r.timestamp
                ].map(val => `"${String(val).replace(/"/g, '""')}"`).join(',');
                csvContent += row + "\r\n";
            });
        });

        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.download = `trip_registrations_${recFilter}.csv`;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }
